# JARVIS AI Assistant - Complete Technical Specification Document

## Project Overview

**Project Name:** JARVIS - Personal AI Assistant  
**Objective:** Create an advanced, always-running AI assistant similar to Marvel's JARVIS/FRIDAY that can:
- Control the computer through voice commands
- Learn and remember user preferences over time
- Access and manage data from multiple sources (phone, calendar, email, etc.)
- Wake up on voice command ("Jarvis") without manual script execution
- Provide intelligent, context-aware responses using RAG (Retrieval-Augmented Generation)

### 2.1 Wake Word Detection Service

**Purpose:** Always-running background service that listens for "Jarvis" wake word

**Requirements:**
- Must consume minimal resources (< 50MB RAM, < 5% CPU)
- Should start automatically on system boot
- Should restart automatically if crashed
- Must work even when other applications are in focus

**Implementation Options:**

**Option A: Picovoice Porcupine (Recommended)**
```python
import pvporcupine
import pyaudio
import struct

class WakeWordDetector:
    def __init__(self):
        self.porcupine = pvporcupine.create(
            access_key='YOUR_PICOVOICE_KEY',
            keywords=['jarvis']  # or use custom .ppn file
        )
        self.audio_stream = None
        
    def start_listening(self):
        """Continuously listen for wake word"""
        pa = pyaudio.PyAudio()
        self.audio_stream = pa.open(
            rate=self.porcupine.sample_rate,
            channels=1,
            format=pyaudio.paInt16,
            input=True,
            frames_per_buffer=self.porcupine.frame_length
        )
        
        while True:
            pcm = self.audio_stream.read(self.porcupine.frame_length)
            pcm = struct.unpack_from("h" * self.porcupine.frame_length, pcm)
            
            keyword_index = self.porcupine.process(pcm)
            
            if keyword_index >= 0:
                self.on_wake_word_detected()
    
    def on_wake_word_detected(self):
        """Trigger main Jarvis system"""
        print("Wake word detected! Activating Jarvis...")
        # Activate main system here
        activate_jarvis_core()
```

### 2.2 Speech Recognition & Processing

**Speech-to-Text:**
```python
import speech_recognition as sr

class SpeechRecognizer:
    def __init__(self):
        self.recognizer = sr.Recognizer()
        self.microphone = sr.Microphone()
        
    def listen_for_command(self, timeout=5):
        """Listen and convert speech to text"""
        with self.microphone as source:
            # Adjust for ambient noise
            self.recognizer.adjust_for_ambient_noise(source, duration=0.5)
            
            print("Listening for command...")
            audio = self.recognizer.listen(source, timeout=timeout)
            
        try:
            # Use Google Speech Recognition (free)
            text = self.recognizer.recognize_google(audio)
            return text
        except sr.UnknownValueError:
            return None
        except sr.RequestError as e:
            print(f"Error: {e}")
            return None
```

**Text-to-Speech:**
```python
import pyttsx3

class TextToSpeech:
    def __init__(self):
        self.engine = pyttsx3.init()
        self.engine.setProperty('rate', 175)  # Speed
        self.engine.setProperty('volume', 0.9)
        
        # Set voice (optional - choose male/female)
        voices = self.engine.getProperty('voices')
        self.engine.setProperty('voice', voices[0].id)
    
    def speak(self, text):
        """Convert text to speech"""
        self.engine.say(text)
        self.engine.runAndWait()
```

```sql
CREATE TABLE conversations (
    id SERIAL PRIMARY KEY,
    timestamp TIMESTAMP DEFAULT NOW(),
    user_input TEXT NOT NULL,
    ai_response TEXT NOT NULL,
    embedding vector(1536),  -- Embedding dimension
    intent VARCHAR(100),      -- Classified intent
    context_used JSONB,       -- What context was retrieved
    action_taken TEXT,        -- What action was executed
    success BOOLEAN,          -- Whether action succeeded
    execution_time_ms INTEGER -- Response time
);

-- Index for vector similarity search
CREATE INDEX ON conversations USING ivfflat (embedding vector_cosine_ops);

-- ============================================

    id SERIAL PRIMARY KEY,
```

### 2.4 RAG System Implementation

**Purpose:** Retrieve relevant context from past conversations to provide intelligent, context-aware responses

**Core RAG Class:**

```python
from groq import Groq
import psycopg2
from pgvector.psycopg2 import register_vector
import numpy as np
from typing import List, Dict
import json

class JarvisRAG:
    def __init__(self, groq_api_key: str, db_config: Dict):
        """
        Initialize RAG system
        
        Args:
            groq_api_key: Your Groq API key
            db_config: PostgreSQL connection config
        """
        self.groq_client = Groq(api_key=groq_api_key)
        self.db = psycopg2.connect(**db_config)
        register_vector(self.db)
        
        # Models
        self.chat_model = "mixtral-8x7b-32768"  # or "llama-3.1-70b-versatile"
        self.embedding_model = "text-embedding-ada-002"  # Configure based on availability
    
    def get_embedding(self, text: str) -> List[float]:
        """
        Generate embedding vector for text
        
        Args:
            text: Input text to embed
            
        Returns:
            List of floats representing the embedding
        """
        # Note: Groq may not have embeddings API
        # Use OpenAI or another service for embeddings
        from openai import OpenAI
        client = OpenAI(api_key="your_openai_key")
        
        response = client.embeddings.create(
            model=self.embedding_model,
            input=text
        )
        return response.data[0].embedding
    
    def retrieve_relevant_context(self, query: str, limit: int = 5) -> List[Dict]:
        """
        Retrieve most relevant past conversations
        
        Args:
            query: User's current query
            limit: Number of results to return
            
        Returns:
            List of relevant conversation snippets
        """
        query_embedding = self.get_embedding(query)
        
        cursor = self.db.cursor()
        cursor.execute("""
            SELECT 
                user_input,
                ai_response,
                timestamp,
                intent,
                action_taken,
                1 - (embedding <=> %s::vector) as similarity
            FROM conversations
            WHERE embedding IS NOT NULL
            ORDER BY embedding <=> %s::vector
            LIMIT %s
        """, (query_embedding, query_embedding, limit))
        
        results = cursor.fetchall()
        cursor.close()
        
        context = []
        for row in results:
            context.append({
                'user_input': row[0],
                'ai_response': row[1],
                'timestamp': row[2].isoformat(),
                'intent': row[3],
                'action_taken': row[4],
                'similarity': row[5]
            })
        
        return context
    
    def get_user_preferences(self, category: str = None) -> Dict:
        """
        Retrieve user preferences from database
        
        Args:
            category: Optional category filter
            
        Returns:
            Dictionary of user preferences
        """
        cursor = self.db.cursor()
        
        if category:
            cursor.execute("""
                SELECT key, value, confidence_score
                FROM user_profile
                WHERE category = %s AND confidence_score > 0.7
                ORDER BY confidence_score DESC
            """, (category,))
        else:
            cursor.execute("""
                SELECT category, key, value, confidence_score
                FROM user_profile
                WHERE confidence_score > 0.7
                ORDER BY confidence_score DESC
                LIMIT 20
            """)
        
        results = cursor.fetchall()
        cursor.close()
        
        preferences = {}
        for row in results:
            if category:
                preferences[row[0]] = {
                    'value': row[1],
                    'confidence': row[2]
                }
            else:
                if row[0] not in preferences:
                    preferences[row[0]] = {}
                preferences[row[0]][row[1]] = {
                    'value': row[2],
                    'confidence': row[3]
                }
        
        return preferences
    
    def build_system_prompt(self, user_preferences: Dict) -> str:
        """
        Build system prompt with user context
        
        Args:
            user_preferences: User's learned preferences
            
        Returns:
            System prompt string
        """
        prefs_str = json.dumps(user_preferences, indent=2)
        
        return f"""You are Jarvis, an advanced personal AI assistant similar to the AI from Iron Man.

Your capabilities:
- Control the computer (open applications, execute commands)
- Access calendar, contacts, and emails
- Remember past conversations and learn user preferences
- Provide proactive suggestions based on patterns
- Respond with intelligence, wit, and efficiency

User Profile & Preferences:
{prefs_str}

Guidelines:
- Be concise but helpful
- Use the user's name when you know it
- Reference past conversations when relevant
- Anticipate needs based on patterns
- Confirm before executing destructive actions
- Maintain a professional but friendly tone"""
    
    def process_command(self, user_input: str) -> Dict:
        """
        Main processing function - handles user command with RAG
        
        Args:
            user_input: User's voice/text input
            
        Returns:
            Dictionary with response and metadata
        """
        # 1. Retrieve relevant past context
        relevant_context = self.retrieve_relevant_context(user_input, limit=5)
        
        # 2. Get user preferences
        user_prefs = self.get_user_preferences()
        
        # 3. Build context string
        context_str = ""
        if relevant_context:
            context_str = "Relevant past interactions:\n"
            for i, ctx in enumerate(relevant_context, 1):
                context_str += f"{i}. [{ctx['timestamp']}] User: {ctx['user_input']}\n"
                context_str += f"   Jarvis: {ctx['ai_response']}\n"
                if ctx['action_taken']:
                    context_str += f"   Action: {ctx['action_taken']}\n"
                context_str += "\n"
        
        # 4. Build system prompt
        system_prompt = self.build_system_prompt(user_prefs)
        
        # 5. Create messages for Groq
        messages = [
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": f"{context_str}\n\nCurrent request: {user_input}"}
        ]
        
        # 6. Get response from Groq
        try:
            response = self.groq_client.chat.completions.create(
                model=self.chat_model,
                messages=messages,
                temperature=0.7,
                max_tokens=500
            )
            
            ai_response = response.choices[0].message.content
            
            # 7. Classify intent (you can use another Groq call or rule-based)
            intent = self.classify_intent(user_input, ai_response)
            
            # 8. Execute action if needed
            action_result = self.execute_action(intent, user_input, ai_response)
            
            # 9. Store interaction in database
            self.store_interaction(
                user_input=user_input,
                ai_response=ai_response,
                intent=intent,
                action_taken=action_result.get('action'),
                success=action_result.get('success')
            )
            
            return {
                'response': ai_response,
                'intent': intent,
                'action_result': action_result,
                'context_used': len(relevant_context)
            }
            
        except Exception as e:
            return {
                'response': f"I encountered an error: {str(e)}",
                'intent': 'error',
                'action_result': {'success': False, 'error': str(e)},
                'context_used': 0
            }
    
    def classify_intent(self, user_input: str, ai_response: str) -> str:
        """
        Classify the intent of user's command
        
        Args:
            user_input: User's input
            ai_response: AI's response
            
        Returns:
            Intent classification string
        """
        # Simple rule-based classification
        # You can make this more sophisticated with ML
        
        user_lower = user_input.lower()
        
        intents = {
            'open_app': ['open', 'launch', 'start', 'run'],
            'close_app': ['close', 'quit', 'exit', 'kill'],
            'search_web': ['search', 'google', 'look up', 'find'],
            'send_email': ['send email', 'email', 'compose'],
            'calendar': ['calendar', 'schedule', 'meeting', 'appointment'],
            'reminder': ['remind', 'reminder', 'remember'],
            'weather': ['weather', 'temperature', 'forecast'],
            'time': ['time', 'what time', 'clock'],
            'general_query': []  # Default
        }
        
        for intent, keywords in intents.items():
            if any(keyword in user_lower for keyword in keywords):
                return intent
        
        return 'general_query'
    
    def execute_action(self, intent: str, user_input: str, ai_response: str) -> Dict:
        """
        Execute system actions based on intent
        
        Args:
            intent: Classified intent
            user_input: Original user input
            ai_response: AI's response
            
        Returns:
            Dictionary with action result
        """
        try:
            if intent == 'open_app':
                # Extract app name (simple extraction)
                import re
                match = re.search(r'open\s+(\w+)', user_input.lower())
                if match:
                    app_name = match.group(1)
                    return self.open_application(app_name)
            
            elif intent == 'search_web':
                # Extract search query
                import re
                match = re.search(r'search\s+(?:for\s+)?(.+)', user_input.lower())
                if match:
                    query = match.group(1)
                    return self.search_web(query)
            
            # Add more intent handlers here
            
            return {'action': None, 'success': True}
            
        except Exception as e:
            return {'action': intent, 'success': False, 'error': str(e)}
    
    def open_application(self, app_name: str) -> Dict:
        """
        Open an application
        
        Args:
            app_name: Name of application
            
        Returns:
            Result dictionary
        """
        import subprocess
        import platform
        
        # Map common names to actual executables
        app_map = {
            'chrome': 'chrome' if platform.system() != 'Windows' else 'chrome.exe',
            'firefox': 'firefox',
            'notepad': 'notepad.exe',
            'calculator': 'calc.exe',
            'spotify': 'spotify',
            # Add more applications
        }
        
        executable = app_map.get(app_name.lower(), app_name)
        
        try:
            subprocess.Popen([executable])
            return {
                'action': f'open_app:{app_name}',
                'success': True,
                'message': f'Opened {app_name}'
            }
        except Exception as e:
            return {
                'action': f'open_app:{app_name}',
                'success': False,
                'error': str(e)
            }
    
    def search_web(self, query: str) -> Dict:
        """
        Perform web search
        
        Args:
            query: Search query
            
        Returns:
            Result dictionary
        """
        import webbrowser
        
        try:
            url = f"https://www.google.com/search?q={query.replace(' ', '+')}"
            webbrowser.open(url)
            return {
                'action': f'search_web:{query}',
                'success': True,
                'message': f'Searching for: {query}'
            }
        except Exception as e:
            return {
                'action': f'search_web:{query}',
                'success': False,
                'error': str(e)
            }
    
    def store_interaction(self, user_input: str, ai_response: str, 
                         intent: str, action_taken: str = None, 
                         success: bool = True):
        """
        Store conversation in database with embedding
        
        Args:
            user_input: User's input
            ai_response: AI's response
            intent: Classified intent
            action_taken: What action was executed
            success: Whether action succeeded
        """
        # Generate embedding
        embedding = self.get_embedding(user_input)
        
        cursor = self.db.cursor()
        cursor.execute("""
            INSERT INTO conversations 
            (user_input, ai_response, embedding, intent, action_taken, success)
            VALUES (%s, %s, %s, %s, %s, %s)
        """, (user_input, ai_response, embedding, intent, action_taken, success))
        
        self.db.commit()
        cursor.close()
    
    def learn_preference(self, category: str, key: str, value: any, 
                        increment_frequency: bool = True):
        """
        Store or update a learned user preference
        
        Args:
            category: Preference category
            key: Preference key
            value: Preference value
            increment_frequency: Whether to increment frequency counter
        """
        cursor = self.db.cursor()
        
        # Convert value to JSON if not already
        if not isinstance(value, str):
            import json
            value = json.dumps(value)
        
        cursor.execute("""
            INSERT INTO user_profile (category, key, value, frequency, confidence_score)
            VALUES (%s, %s, %s, 1, 0.5)
            ON CONFLICT (category, key) 
            DO UPDATE SET
                value = EXCLUDED.value,
                frequency = user_profile.frequency + CASE WHEN %s THEN 1 ELSE 0 END,
                confidence_score = LEAST(1.0, user_profile.confidence_score + 0.1),
                last_updated = NOW()
        """, (category, key, value, increment_frequency))
        
        self.db.commit()
        cursor.close()
```

```python
class PhoneAPI:

        contacts = self.get_contacts()
        recent_calls = self.get_recent_calls()
        notifications = self.get_notifications()

                VALUES ('phone', 'contact', %s)

            """, (json.dumps(contact),))
        
        # Store calls
        for call in recent_calls:
            cursor.execute("""
                INSERT INTO external_data (source, data_type, content)
                VALUES ('phone', 'call', %s)
            """, (json.dumps(call),))
        
        # Store notifications
        for notification in notifications:
            cursor.execute("""
                INSERT INTO external_data (source, data_type, content)
                VALUES ('phone', 'notification', %s)
            """, (json.dumps(notification),))
```

```python
        print("Jarvis activated!")
        
        # Start listening for command
        asyncio.run(self.listen_and_process())
    
    async def listen_and_process(self):
        """Main command processing loop"""
        while self.is_active:
            # Listen for user command
            user_input = self.speech_recognizer.listen_for_command(timeout=10)
            
            if user_input is None:
                # No command detected, go back to sleep
                self.tts.speak("Going back to sleep mode")
                self.is_active = False
                break
            
            print(f"User said: {user_input}")
            
            # Check for exit commands
            if any(word in user_input.lower() for word in ['sleep', 'standby', 'goodbye']):
                self.tts.speak("Going to sleep mode. Call me when you need me.")
                self.is_active = False
                break
            
            # Process command with RAG
            result = self.rag.process_command(user_input)
            
            # Speak response
            self.tts.speak(result['response'])
            
            # Execute any required actions
            if result.get('action_result'):
                action = result['action_result']
                if not action.get('success'):
                    self.tts.speak(f"I encountered an issue: {action.get('error', 'Unknown error')}")
            
            # Continue listening for next command
            await asyncio.sleep(0.1)
    
    def deactivate(self):
        """Deactivate Jarvis"""
        self.is_active = False
        print("Jarvis deactivated")
```

```python
from datetime import datetime, timedelta
from collections import Counter
import json

class LearningEngine:
    def __init__(self, db_connection):
        self.db = db_connection
    
    def analyze_usage_patterns(self):
        """Analyze user behavior and detect patterns"""
        
        # 1. Time-based patterns
        time_patterns = self.detect_time_patterns()
        
        # 2. Frequently used commands
        frequent_commands = self.detect_frequent_commands()
        
        # 3. Application usage patterns
        app_patterns = self.detect_app_patterns()
        
        # 4. Contextual patterns
        context_patterns = self.detect_contextual_patterns()
        
        return {
            'time_patterns': time_patterns,
            'frequent_commands': frequent_commands,
            'app_patterns': app_patterns,
            'context_patterns': context_patterns
        }
    
    def detect_time_patterns(self) -> Dict:
        """Detect time-based usage patterns"""
        cursor = self.db.cursor()
        
        # Get actions from last 30 days
        cursor.execute("""
            SELECT 
                EXTRACT(HOUR FROM timestamp) as hour,
                EXTRACT(DOW FROM timestamp) as day_of_week,
                action_taken,
                COUNT(*) as frequency
            FROM conversations
            WHERE 
                timestamp > NOW() - INTERVAL '30 days'
                AND action_taken IS NOT NULL
            GROUP BY hour, day_of_week, action_taken
            HAVING COUNT(*) > 3
            ORDER BY frequency DESC
        """)
        
        patterns = []
        for row in cursor.fetchall():
            hour, day_of_week, action, frequency = row
            
            # Convert day_of_week (0=Sunday) to name
            days = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 
                   'Thursday', 'Friday', 'Saturday']
            
            patterns.append({
                'hour': int(hour),
                'day': days[int(day_of_week)],
                'action': action,
                'frequency': frequency,
                'confidence': min(0.95, frequency / 30)  # Max 95% confidence
            })
        
        cursor.close()
        return patterns
    
    def detect_frequent_commands(self, min_frequency: int = 5) -> List[Dict]:
        """Detect frequently used commands"""
        cursor = self.db.cursor()
        
        cursor.execute("""
            SELECT 
                intent,
                action_taken,
                COUNT(*) as frequency,
                AVG(CASE WHEN success THEN 1.0 ELSE 0.0 END) as success_rate
            FROM conversations
            WHERE timestamp > NOW() - INTERVAL '30 days'
            GROUP BY intent, action_taken
            HAVING COUNT(*) >= %s
            ORDER BY frequency DESC
            LIMIT 20
        """, (min_frequency,))
        
        commands = []
        for row in cursor.fetchall():
            intent, action, frequency, success_rate = row
            commands.append({
                'intent': intent,
                'action': action,
                'frequency': frequency,
                'success_rate': float(success_rate),
                'suggestion': self.generate_automation_suggestion(intent, action, frequency)
            })
        
        cursor.close()
        return commands
    
    def detect_app_patterns(self) -> Dict:
        """Detect application usage patterns"""
        cursor = self.db.cursor()
        
        cursor.execute("""
            SELECT 
                action_taken,
                COUNT(*) as frequency,
                ARRAY_AGG(DISTINCT EXTRACT(HOUR FROM timestamp)) as hours_used
            FROM conversations
            WHERE 
                action_taken LIKE 'open_app:%'
                AND timestamp > NOW() - INTERVAL '30 days'
            GROUP BY action_taken
            HAVING COUNT(*) > 3
        """)
        
        app_patterns = {}
        for row in cursor.fetchall():
            action, frequency, hours = row
            app_name = action.split(':')[1]
            
            app_patterns[app_name] = {
                'frequency': frequency,
                'typical_hours': [int(h) for h in hours],
                'suggestion': self.suggest_app_automation(app_name, hours, frequency)
            }
        
        cursor.close()
        return app_patterns
    
    def generate_automation_suggestion(self, intent: str, 
                                      action: str, frequency: int) -> str:
        """Generate automation suggestions based on patterns"""
        if frequency > 20:
            return f"I notice you use '{action}' very frequently. Would you like me to create a shortcut?"
        elif frequency > 10:
            return f"You often use '{action}'. I can automate this if you'd like."
        return None
    
    def suggest_app_automation(self, app_name: str, 
                              hours: List[int], frequency: int) -> str:
        """Suggest app opening automation"""
        if len(set(hours)) <= 2 and frequency > 15:
            # User opens this app at consistent times
            typical_hour = Counter(hours).most_common(1)[0][0]
            return f"I notice you typically open {app_name} around {typical_hour}:00. Should I open it automatically?"
        return None
    
    def learn_from_feedback(self, conversation_id: int, 
                          feedback: str, notes: str = None):
        """
        Learn from user feedback
        
        Args:
            conversation_id: ID of the conversation
            feedback: 'positive', 'negative', or 'neutral'
            notes: Optional notes about the feedback
        """
        cursor = self.db.cursor()
        
        # Update action log
        cursor.execute("""
            INSERT INTO action_log (
                timestamp, action_type, parameters, user_feedback
            )
            SELECT 
                timestamp, action_taken, 
                jsonb_build_object('conversation_id', %s, 'notes', %s),
                %s
            FROM conversations
            WHERE id = %s
        """, (conversation_id, notes, feedback, conversation_id))
        
        # If negative feedback, reduce confidence in similar patterns
        if feedback == 'negative':
            cursor.execute("""
                SELECT intent, action_taken FROM conversations WHERE id = %s
            """, (conversation_id,))
            
            result = cursor.fetchone()
            if result:
                intent, action = result
                
                # Reduce confidence in user profile for similar patterns
                cursor.execute("""
                    UPDATE user_profile
                    SET confidence_score = GREATEST(0.1, confidence_score - 0.2)
                    WHERE value::text LIKE %s
                """, (f'%{action}%',))
        
        self.db.commit()
        cursor.close()
    
    def get_proactive_suggestions(self) -> List[str]:
        """
        Generate proactive suggestions based on patterns and context
        
        Returns:
            List of suggestion strings
        """
        patterns = self.analyze_usage_patterns()
        suggestions = []
        
        current_hour = datetime.now().hour
        current_day = datetime.now().strftime('%A')
        
        # Check time patterns
        for pattern in patterns['time_patterns']:
            if (pattern['hour'] == current_hour and 
                pattern['day'] == current_day and 
                pattern['confidence'] > 0.7):
                
                suggestions.append(
                    f"It's {current_hour}:00. You usually {pattern['action']} at this time. "
                    "Would you like me to do that now?"
                )
        
        # Check app patterns
        for app, data in patterns['app_patterns'].items():
            if current_hour in data['typical_hours'] and data['frequency'] > 15:
                suggestions.append(
                    f"You typically use {app} around this time. Should I open it?"
                )
        
        return suggestions[:3]  # Return top 3 suggestions
```

```python
import schedule

class ProactiveNotifier:
    def __init__(self, jarvis_core: JarvisCore, learning_engine: LearningEngine):

        self.learning = learning_engine
        self.running = False
    
    def start(self):
        """Start proactive notification system"""
        self.running = True
        
        # Schedule checks
        schedule.every(30).minutes.do(self.check_suggestions)
        schedule.every(1).hours.do(self.check_calendar)
        schedule.every(1).day.at("09:00").do(self.morning_briefing)
        
        # Run in separate thread
        thread = Thread(target=self._run_scheduler)
        thread.daemon = True
        thread.start()
    
    def _run_scheduler(self):
        """Run the scheduler loop"""
        while self.running:
            schedule.run_pending()
            time.sleep(60)
    
    def check_suggestions(self):
        """Check for proactive suggestions"""
        suggestions = self.learning.get_proactive_suggestions()
        
        if suggestions and not self.jarvis.is_active:
            # Activate Jarvis to give suggestion
            self.jarvis.tts.speak(suggestions[0])

        """Check upcoming calendar events"""
        if self.jarvis.google:
            events = self.jarvis.google.get_calendar_events(days=1)
            
            # Check for events in next hour

            now = datetime.now()
            soon = now + timedelta(hours=1)
            
            for event in events:
                event_time = datetime.fromisoformat(
                    event['start'].get('dateTime', event['start'].get('date'))
                )
                
                if now < event_time < soon:
                    self.jarvis.tts.speak(
                        f"Reminder: You have {event['summary']} "
                        f"in {int((event_time - now).seconds / 60)} minutes"
                    )
    
    def morning_briefing(self):
```

```python
                topics.append('scheduling')
            if 'email' in text or 'message' in text:
                topics.append('communication')
            if 'file' in text or 'document' in text:
                topics.append('file_management')
        
        self.update_context('topics', list(set(topics)))
    
    def should_ask_for_clarification(self, user_input: str) -> bool:
        """Determine if clarification is needed"""
        # Check if input is ambiguous
        ambiguous_keywords = ['it', 'that', 'this', 'them', 'they']
        
        words = user_input.lower().split()
        has_ambiguous = any(word in words for word in ambiguous_keywords)
        has_context = len(self.current_context) > 0
        
        return has_ambiguous and not has_context
```

## 9. Troubleshooting Guide

**Common Issues:**

**Issue 1: Wake word not detected**
- Check microphone permissions
- Verify Picovoice API key is valid
- Test microphone with other applications
- Adjust sensitivity settings in wake word detector
- Check if service is running: `systemctl status jarvis` (Linux) or Task Manager (Windows)

**Issue 2: Database connection errors**
- Verify PostgreSQL is running
- Check connection credentials in config.json
- Ensure pgvector extension is installed: `CREATE EXTENSION vector;`
- Check firewall settings if using remote database

**Issue 3: Speech recognition not working**
- Check internet connection (Google Speech Recognition requires internet)
- Verify microphone input levels
- Test with: `python -m speech_recognition`
- Consider switching to offline model like Vosk

**Issue 4: Groq API errors**
- Verify API key is valid
- Check API rate limits
- Ensure you have credits/quota remaining
- Test with simple API call outside the application

**Issue 5: High CPU/memory usage**
- Wake word detection consuming too much: Try different sensitivity settings
- Database queries slow: Add indexes, optimize queries
- Too many background processes: Adjust polling intervals
- Memory leaks: Restart service periodically

**Issue 6: Text-to-Speech not working**
- Check if pyttsx3 is properly installed
- Try alternative TTS: `pip install gTTS`
- Verify audio output device is working
- Check volume settings

## 16. Backup & Recovery

### Database Backup

```bash
#!/bin/bash
# backup_db.sh

BACKUP_DIR="/path/to/backups"
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_FILE="$BACKUP_DIR/jarvis_backup_$DATE.sql"

# Create backup
pg_dump jarvis_db > $BACKUP_FILE

# Compress
gzip $BACKUP_FILE

# Keep only last 7 days of backups
find $BACKUP_DIR -name "jarvis_backup_*.sql.gz" -mtime +7 -delete

echo "Backup completed: $BACKUP_FILE.gz"
```

### Restore from Backup

```bash
# Uncompress
gunzip jarvis_backup_20240108_120000.sql.gz

# Restore
psql jarvis_db < jarvis_backup_20240108_120000.sql
```

## 18. Privacy & Data Management

### Data Collection

Jarvis collects and stores:
- Voice commands (converted to text)
- User interactions and preferences
- System actions and results
- Time-based patterns
- External data (contacts, calendar, etc.)

### Data Retention

- **Conversations**: Kept indefinitely for learning (can be configured)
- **External data cache**: Refreshed every 24 hours
- **Logs**: Rotated after 10MB, keep last 5 files
- **Backups**: Keep last 30 days

### Privacy Controls

```python
class PrivacyManager:
    def __init__(self, db_connection):
        self.db = db_connection
    
    def clear_conversations(self, before_date=None):
        """Clear conversation history"""
        cursor = self.db.cursor()
        
        if before_date:
            cursor.execute("""
                DELETE FROM conversations 
                WHERE timestamp < %s
            """, (before_date,))
        else:
            cursor.execute("DELETE FROM conversations")
        
        self.db.commit()
        cursor.close()
    
    def export_user_data(self, output_file: str):
        """Export all user data"""
        cursor = self.db.cursor()
        
        # Export conversations
        cursor.execute("SELECT * FROM conversations")
        conversations = cursor.fetchall()
        
        # Export preferences
        cursor.execute("SELECT * FROM user_profile")
        preferences = cursor.fetchall()
        
        data = {
            'conversations': conversations,
            'preferences': preferences,
            'exported_at': datetime.now().isoformat()
        }
        
        with open(output_file, 'w') as f:
            json.dump(data, f, indent=2, default=str)
        
        cursor.close()
    
    def anonymize_data(self):
        """Anonymize personal data"""
        cursor = self.db.cursor()
        
        # Remove names and personal identifiers
        cursor.execute("""
            UPDATE conversations
            SET user_input = regexp_replace(user_input, '[A-Z][a-z]+ [A-Z][a-z]+', '[NAME]', 'g'),
                ai_response = regexp_replace(ai_response, '[A-Z][a-z]+ [A-Z][a-z]+', '[NAME]', 'g')
        """)
        
        # Clear external data
        cursor.execute("DELETE FROM external_data WHERE source = 'phone'")
        
        self.db.commit()
        cursor.close()
```

```python
from src.core.jarvis_core import JarvisCore
```
